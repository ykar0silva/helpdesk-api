# --- Estágio 1: BUILD (Compilação) ---
# Usa uma imagem que já tem o Maven e o Java 17 instalados
FROM maven:3.8.5-openjdk-17 AS build
WORKDIR /app

# 1. Copia só o pom.xml primeiro para baixar as dependências (cache inteligente)
COPY pom.xml .
RUN mvn dependency:go-offline

# 2. Copia o resto do código e compila o projeto
COPY src ./src
RUN mvn clean package -DskipTests

# --- Estágio 2: RUNTIME (Execução) ---
# Usa uma imagem muito leve (slim) só para rodar o Java
FROM openjdk:17-jdk-slim
WORKDIR /app

# 3. Pega o arquivo .jar que foi gerado no estágio anterior
COPY --from=build /app/target/*.jar app.jar

# 4. Cria a pasta de uploads dentro do container
RUN mkdir -p /app/uploads

# 5. Expõe a porta e roda
EXPOSE 8080
ENTRYPOINT ["java", "-jar", "app.jar"]
